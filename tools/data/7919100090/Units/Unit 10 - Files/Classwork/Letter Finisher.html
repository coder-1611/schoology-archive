recipient = input('Who are you sending a letter to? ')
sender = input('Who is sending the letter? ')
filename = input('Which file contains the letter? ')

# step 1
f = open(filename)
contents = f.read()
f.close()

# step 2
letter = f'''Dear {recipient},

{contents}

Love,
{sender}'''

# step 3
f = open(filename, 'w')
f.write(letter)
f.close()
