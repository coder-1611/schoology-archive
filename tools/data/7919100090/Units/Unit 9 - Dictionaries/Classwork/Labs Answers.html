def display_inventory(inventory):
    print('Inventory:')
    item_total = 0 # Initialize the counter
    for item, count in inventory.items():
        print(count, item) # Print out each item in the correct format
        item_total += count # Increment the counter
    print('Total number of items:', item_total)
